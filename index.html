class HalloweenGame { #score = 0; #state = "ready"; #targets = new Set(); constructor({ totalTiles = 12, targetCount = 4 } = {}) { if ( !Number.isInteger(totalTiles) || !Number.isInteger(targetCount) || totalTiles < 1 || targetCount < 1 || targetCount > totalTiles ) { throw new RangeError("Ogiltiga spelinställningar."); } this.totalTiles = totalTiles; this.targetCount = targetCount; } start() { this.#score = 0; this.#state = "playing"; this.#targets = new Set(); while (this.#targets.size < this.targetCount) { this.#targets.add( Math.floor(Math.random() * this.totalTiles) ); } return this.getSnapshot(); } selectTile(index) { if (this.#state !== "playing") { throw new Error("Spelet är inte aktivt."); } if ( !Number.isInteger(index) || index < 0 || index >= this.totalTiles ) { throw new RangeError("Ogiltig spelruta."); } if (this.#targets.delete(index)) { this.#score += 1; } if (this.#targets.size === 0) { this.#state = "won"; } return this.getSnapshot(); } getSnapshot() { return Object.freeze({ score: this.#score, state: this.#state, remaining: this.#targets.size }); } }
